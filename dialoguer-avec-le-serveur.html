    8. Dialoguer avec le serveur

    Les exemples de ce chapitre interrogent le serveur JSON [RdvMedecins] : il doit être lancé ([npm run start:dev] dans le dossier [rdvmedecins-nestjs-json]), sur le port 8080, en même temps que le serveur de développement d'[Angular], sur le port 4200.

    8.1. Exemple 22 : une première requête HTTP

    8.1.1. Le proxy du serveur de développement

    La page de l'exemple vient de http://localhost:4200 (le serveur de développement d'[Angular], lancé par [ng serve]) ; l'API est sur http://localhost:8080. Pour un navigateur, ce sont deux origines différentes (le port fait partie de l'origine). Or, par sécurité, un navigateur n'autorise pas une page à lire les réponses d'une autre origine, sauf accord explicite du serveur (le mécanisme dit CORS). Et une requête vers une autre origine n'emporte pas les cookies (donc pas le jeton), sauf si le code le demande ([withCredentials: true]) ET que le serveur l'accepte, toujours par CORS.

    Plutôt que d'ouvrir le serveur aux autres origines, on fait passer les requêtes de l'API par le serveur de développement d'[Angular]. C'est le proxy déclaré dans le fichier [proxy.conf.json] de l'espace de travail :

// 22-httpclient/src/app/app.ts - une première requête HTTP vers le serveur
// -------------------------------------------------------------------------
// Le serveur JSON RdvMedecins doit être lancé (port 8080). On lui demande
// un captcha : GET /api/captcha -> { "id": "...", "svg": "<svg ...>" }.
//
//   - HttpClient : le client HTTP d'Angular (fourni d'office à toute
//     l'application). http.get<T>(url) renvoie un OBSERVABLE (cf. exemple
//     19) : la requête ne part qu'au moment où l'on s'y ABONNE (subscribe) ;
//     la réponse JSON arrive déjà convertie en objet, du type T annoncé ;
//   - l'adresse est RELATIVE (/api/captcha) : la requête part vers le
//     serveur de la page, celui d'Angular (port 4200), qui la transmet au
//     serveur JSON (port 8080) grâce à son proxy (cf. proxy.conf.json) ;
//   - trois états à afficher : en cours, réussi, en erreur.
// =========================================================================

import { Component, computed, inject, signal } from '@angular/core';
import { HttpClient, HttpErrorResponse } from '@angular/common/http';
import { DomSanitizer, SafeHtml } from '@angular/platform-browser';

// ce que renvoie GET /api/captcha
interface Captcha {
  id: string;
  svg: string;
}

@Component({
  selector: 'app-root',
  templateUrl: './app.html',
  styles: `
    .captcha {
      display: inline-block;
      border: 1px solid #dee2e6;
      border-radius: 0.375rem;
    }
  `,
})
export class App {
  private readonly http = inject(HttpClient);
  private readonly sanitizer = inject(DomSanitizer);

  protected readonly captcha = signal<Captcha | null>(null);
  protected readonly enCours = signal(false);
  protected readonly erreur = signal<string | null>(null);
  protected readonly nbRequetes = signal(0);

  // [innerHTML] insère une chaîne comme du HTML. Par sécurité, Angular
  // NETTOIE ce HTML ("sanitization") : il en retire les scripts... et les
  // balises <svg>. bypassSecurityTrustHtml dit à Angular : « ce HTML est
  // sûr, ne le nettoie pas ». À n'utiliser QUE pour un contenu sûr : ce SVG
  // est fabriqué par NOTRE serveur. Avec une saisie d'utilisateur, ce serait
  // ouvrir la porte à l'injection de scripts ({{ }} échappe, lui, tout).
  protected readonly svg = computed<SafeHtml | null>(() => {
    const c = this.captcha();
    return c ? this.sanitizer.bypassSecurityTrustHtml(c.svg) : null;
  });

  constructor() {
    this.charger(); // au démarrage, on charge un captcha
  }

  protected charger(): void {
    this.enCours.set(true);
    this.erreur.set(null);
    this.http.get<Captcha>('/api/captcha').subscribe({
      // next : la réponse (code 2xx), déjà convertie depuis le JSON
      next: (captcha) => {
        this.captcha.set(captcha);
        this.nbRequetes.update((n) => n + 1);
        this.enCours.set(false);
      },
      // error : une réponse 4xx / 5xx, ou pas de réponse du tout (status 0)
      error: (e: HttpErrorResponse) => {
        this.erreur.set(`le serveur a répondu ${e.status} ${e.statusText}`);
        this.captcha.set(null);
        this.enCours.set(false);
      },
    });
  }
}

    • lignes 42-45 : les trois états d'une requête : en cours, réussie (le captcha), en erreur ;
    • ligne 65 : [http.get<Captcha>(url)] prépare une requête GET. Elle renvoie un observable (exemple 19) : la requête ne part qu'au moment où l'on s'y abonne ([subscribe]). L'adresse est relative : la requête part vers le serveur de la page, qui la transmet au serveur JSON ;
    • lignes 67-71 : [next] reçoit la réponse, déjà convertie depuis le JSON. [<Captcha>] affirme à [TypeScript] la forme de l'objet : le compilateur ne peut pas vérifier ce qu'enverra le serveur ;
    • lignes 73-77 : [error] reçoit une [HttpErrorResponse], pour toute réponse 4xx ou 5xx, et aussi quand aucune réponse n'arrive (serveur arrêté, réseau coupé : [status] vaut alors 0). Si le serveur JSON est arrêté, c'est le proxy qui répond, par une erreur 5xx ;
    • ligne 59 : au démarrage, on charge un captcha ;
    • lignes 53-56 : l'image SVG du captcha est une chaîne de caractères, qu'il faut insérer comme du HTML ([[innerHTML]] dans le template). Par sécurité, [Angular] nettoie tout HTML inséré ainsi : il en retire les scripts... et les balises [<svg>]. [bypassSecurityTrustHtml] lui indique que ce HTML est sûr. On ne l'utilise QUE pour un contenu sûr : ce SVG est fabriqué par NOTRE serveur. Avec une donnée saisie par un utilisateur, ce serait permettre l'injection de scripts dans la page ; les doubles accolades [{{ }}], elles, échappent tout.

// 23-connexion/src/app/app.ts - se connecter au serveur RdvMedecins
// -------------------------------------------------------------------------
//   1. au démarrage : GET /api/auth/moi - sommes-nous déjà connectés ?
//      (après un F5, le client a tout oublié ; le serveur, lui, lit le cookie) ;
//   2. connexion : POST /api/auth/login, corps JSON { login, password,
//      captchaId, captcha }. En cas de succès, le serveur dépose le cookie
//      httpOnly du jeton JWT : notre code ne le voit JAMAIS (essayez
//      document.cookie dans la console : il n'y est pas) ;
//   3. requête protégée : GET /api/medecins. Le navigateur y joint le cookie
//      tout seul (même origine) ;
//   4. déconnexion : POST /api/auth/logout - le serveur efface le cookie.
//
// firstValueFrom(observable) : une PROMESSE de la première valeur du flux.
// On peut alors écrire les enchaînements de requêtes avec async / await,
// plus lisibles qu'une cascade de subscribe. Une réponse d'erreur rejette
// la promesse avec une HttpErrorResponse : un simple try / catch.
//
// Un cookie appartient à une machine (localhost) et à un chemin (/api) : il
// est donc commun à TOUS les exemples. Une fois connecté ici, les exemples
// 24 et 25 le sont aussi.
// =========================================================================

import { Component, computed, inject, signal } from '@angular/core';
import { HttpClient, HttpErrorResponse } from '@angular/common/http';
import { DomSanitizer } from '@angular/platform-browser';
import { FormsModule } from '@angular/forms';
import { firstValueFrom } from 'rxjs';

interface Utilisateur {
  login: string;
  nom: string;
  role: 'ADMIN' | 'DOCTOR' | 'USER';
}
interface Medecin {
  id: number;
  titre: string;
  prenom: string;
  nom: string;
}
interface Captcha {
  id: string;
  svg: string;
}
// la forme de TOUTES les erreurs de l'API
interface ErreurApi {
  statusCode: number;
  cle: string;
  params?: Record<string, string>;
}

// les quelques messages dont cet exemple a besoin (l'exemple 25 utilisera ngx-translate)
const MESSAGES: Record<string, string> = {
  'LOGIN.ERROR': 'Login ou mot de passe incorrect.',
  'ERRORS.WRONG_CAPTCHA': "Les caractères recopiés ne correspondent pas à l'image.",
  'ERRORS.TOO_MANY_ATTEMPTS': 'Trop de tentatives infructueuses. Réessayez plus tard.',
};

@Component({
  selector: 'app-root',
  imports: [FormsModule],
  templateUrl: './app.html',
})
export class App {
  private readonly http = inject(HttpClient);
  private readonly sanitizer = inject(DomSanitizer);

  protected readonly utilisateur = signal<Utilisateur | null>(null);
  protected readonly medecins = signal<Medecin[]>([]);
  protected readonly captcha = signal<Captcha | null>(null);
  protected readonly svg = computed(() => this.sanitizer.bypassSecurityTrustHtml(this.captcha()?.svg ?? ''));
  protected readonly login = signal('admin');
  protected readonly password = signal('');
  protected readonly reponseCaptcha = signal('');
  protected readonly erreur = signal<string | null>(null);
  protected readonly pret = signal(false); // la vérification de départ est-elle terminée ?

  constructor() {
    this.verifierSession().catch(() => {
      this.erreur.set('Le serveur JSON RdvMedecins est-il lancé (port 8080) ?');
      this.pret.set(true);
    });
  }

  protected async chargerCaptcha(): Promise<void> {
    this.captcha.set(await firstValueFrom(this.http.get<Captcha>('/api/captcha')));
    this.reponseCaptcha.set('');
  }

  // 1. qui est connecté ? 200 -> l'utilisateur ; 401 -> personne
  private async verifierSession(): Promise<void> {
    try {
      this.utilisateur.set(await firstValueFrom(this.http.get<Utilisateur>('/api/auth/moi')));
      await this.chargerMedecins();
    } catch (e) {
      if (!(e instanceof HttpErrorResponse) || e.status !== 401) {
        throw e; // pas un 401 : le serveur ne répond pas
      }
      await this.chargerCaptcha();
    }
    this.pret.set(true);
  }

  // 2. la connexion ; post(url, corps) : le corps est envoyé en JSON
  protected async seConnecter(): Promise<void> {
    this.erreur.set(null);
    try {
      const u = await firstValueFrom(
        this.http.post<Utilisateur>('/api/auth/login', {
          login: this.login(),
          password: this.password(),
          captchaId: this.captcha()?.id,
          captcha: this.reponseCaptcha(),
        }),
      );
      this.utilisateur.set(u);
      this.password.set('');
      await this.chargerMedecins();
    } catch (e) {
      // e.error : le corps JSON de la réponse d'erreur
      const erreurApi = (e as HttpErrorResponse).error as ErreurApi | null;
      const cle = erreurApi?.cle ?? 'ERRORS.SERVER_UNREACHABLE';
      this.erreur.set(`${(e as HttpErrorResponse).status} - ${MESSAGES[cle] ?? cle}`);
      await this.chargerCaptcha(); // chaque image ne sert qu'une fois
    }
  }

  // 3. une requête protégée
  private async chargerMedecins(): Promise<void> {
    this.medecins.set(await firstValueFrom(this.http.get<Medecin[]>('/api/medecins')));
  }

  // 4. la déconnexion
  protected async seDeconnecter(): Promise<void> {
    await firstValueFrom(this.http.post('/api/auth/logout', {}));
    this.utilisateur.set(null);
    this.medecins.set([]);
    await this.chargerCaptcha();
  }
}

// commun/erreur-api.ts - UNE classe pour toutes les erreurs de l'API
// -------------------------------------------------------------------------
// Toutes les réponses d'erreur du serveur RdvMedecins ont la même forme :
//   { "statusCode": 409, "cle": "ERRORS.LOGIN_TAKEN", "params": {...}, "champs": {...} }
// HttpClient les signale par une HttpErrorResponse, dont la propriété error
// contient ce corps JSON... ou autre chose : null (pas de réponse), du texte
// (une page d'erreur du proxy quand le serveur JSON est arrêté). On
// convertit tout cela en UNE classe, ErreurApi, qui porte la clé de
// traduction envoyée par le serveur (et, s'il y en a, ses paramètres et
// l'erreur de chaque champ).
// =========================================================================

import { HttpErrorResponse } from '@angular/common/http';

export class ErreurApi extends Error {
  constructor(
    public readonly statusCode: number, // 0 : le serveur n'a pas répondu
    public readonly cle: string, // ex : 'ERRORS.LOGIN_TAKEN'
    public readonly params: Record<string, string> = {},
    public readonly champs: Record<string, string> = {},
  ) {
    super(cle);
    this.name = 'ErreurApi';
  }
}

// une erreur quelconque -> une ErreurApi
export function versErreurApi(e: unknown): ErreurApi {
  if (e instanceof ErreurApi) {
    return e;
  }
  if (!(e instanceof HttpErrorResponse)) {
    return new ErreurApi(0, 'ERRORS.SERVER_ERROR');
  }
  // le corps JSON de l'erreur, s'il existe et a la bonne forme
  const corps = (typeof e.error === 'object' && e.error !== null ? e.error : {}) as {
    cle?: string;
    params?: Record<string, string>;
    champs?: Record<string, string>;
  };
  // pas de clé : ce n'est pas notre serveur qui a répondu (proxy, réseau)
  const cle = corps.cle ?? (e.status === 0 || e.status >= 500 ? 'ERRORS.SERVER_UNREACHABLE' : 'ERRORS.SERVER_ERROR');
  return new ErreurApi(e.status, cle, corps.params, corps.champs);
}

    await this.nouveauCaptcha();

// -------------------------------------------------------------------------
// Connectez-vous (barre du haut), choisissez un médecin et un jour : l'agenda
// s'affiche. ADMIN et DOCTOR peuvent réserver pour un patient et annuler ;
// un patient (martin...) réserve pour lui-même. Chaque action se termine par
// un rechargement de l'agenda : c'est le SERVEUR qui fait foi.
//
// httpResource : une requête GET décrite par une fonction qui lit des
// signaux (ici idMedecin et jour). Angular l'envoie au départ, puis la
// RENVOIE chaque fois que l'un de ces signaux change. Le résultat est un
// objet de signaux : value() (la réponse), isLoading(), error()... et
// reload() pour la renvoyer à la demande.
// =========================================================================

import { Component, computed, effect, inject, signal } from '@angular/core';
import { httpResource } from '@angular/common/http';
import { FormsModule } from '@angular/forms';
import { BarreConnexion } from '../../../commun/barre-connexion';
import { ErreurApi } from '../../../commun/erreur-api';
import { Agenda, CreneauAgenda, Personne, Utilisateur } from '../../../commun/types';
import { RdvMedecinsService } from './rdvmedecins.service';

function demain(): string {
  const d = new Date(Date.now() + 24 * 3600 * 1000);
  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`;
}

@Component({
  selector: 'app-root',
  imports: [FormsModule, BarreConnexion],
  templateUrl: './app.html',
})
export class App {
  private readonly rdv = inject(RdvMedecinsService);

  protected readonly utilisateur = signal<Utilisateur | null>(null);
  protected readonly medecins = signal<Personne[]>([]);
  protected readonly clients = signal<Personne[]>([]);
  protected readonly idMedecin = signal<number | null>(null);
  protected readonly jour = signal(demain());
  protected readonly idClient = signal<number | null>(null);
  protected readonly message = signal<{ type: 'success' | 'danger'; texte: string } | null>(null);

  protected readonly gestionnaire = computed(() => this.utilisateur() !== null && this.utilisateur()?.role !== 'USER');

  // undefined : pas de requête (aucun médecin choisi, ou personne de connecté)
  protected readonly agenda = httpResource<Agenda>(() => {
    const id = this.idMedecin();
    return id === null || this.utilisateur() === null ? undefined : this.rdv.requeteAgenda(id, this.jour());
  });

  // l'erreur de la dernière requête de l'agenda, sous forme d'ErreurApi
  protected readonly erreurAgenda = computed(() => {
    const e = this.agenda.error();
    return e instanceof ErreurApi ? `${e.statusCode} ${e.cle}` : null;
  });

  constructor() {
    // un nouveau choix de médecin ou de jour efface le message précédent
    effect(() => {
      this.idMedecin();
      this.jour();
      this.message.set(null);
    });
  }

  protected horaire(c: CreneauAgenda): string {
    const hm = (h: number, m: number) => `${h}:${String(m).padStart(2, '0')}`;
    return `${hm(c.hdebut, c.mdebut)} - ${hm(c.hfin, c.mfin)}`;
  }

  // une erreur de l'API devient un message (sa clé, faute de dictionnaire ici)
  private signaler(e: unknown): void {
    this.message.set({ type: 'danger', texte: e instanceof ErreurApi ? `${e.statusCode} ${e.cle}` : String(e) });
  }

  // connexion / déconnexion : on (re)charge ce que ce rôle a le droit de voir
  protected async surConnexion(u: Utilisateur | null): Promise<void> {
    this.utilisateur.set(u);
    this.medecins.set([]);
    this.clients.set([]);
    if (u === null) {
      return;
    }
    try {
      this.medecins.set(await this.rdv.getMedecins());
      // la liste des patients : seulement pour ADMIN et DOCTOR (un USER recevrait 403)
      this.clients.set(this.gestionnaire() ? await this.rdv.getClients() : []);
    } catch (e) {
      this.signaler(e);
    }
  }

  protected async reserver(c: CreneauAgenda): Promise<void> {
    try {
      await this.rdv.reserver(this.jour(), c.id, this.gestionnaire() ? (this.idClient() ?? undefined) : undefined);
      this.message.set({ type: 'success', texte: `Créneau ${this.horaire(c)} réservé.` });
    } catch (e) {
      this.signaler(e);
    }
    this.agenda.reload();
  }

  protected async annuler(c: CreneauAgenda): Promise<void> {
    try {

// 25-erreurs-serveur/src/app/app.ts - afficher les erreurs du SERVEUR
// -------------------------------------------------------------------------
// La gestion des clients (compte admin). Ce formulaire ne fait
// VOLONTAIREMENT aucune validation dans le navigateur : son schéma est vide.
// Tout est envoyé au serveur, qui répond, en cas de refus, par une erreur
// JSON (convertie en ErreurApi par l'intercepteur) :
//
//   400 { cle: 'ERRORS.INVALID_FORM', champs: { nom: 'ERRORS.LASTNAME_REQUIRED' } }
//   409 { cle: 'ERRORS.PATIENT_ALREADY_EXISTS', champs: { prenom: ..., nom: ... } }
//   409 { cle: 'ERRORS.CONCURRENT_UPDATE' }         (verrou optimiste)
//
// submit(formulaire, action) : l'action envoie le modèle au serveur. Si elle
// renvoie des ERREURS (un tableau), les formulaires à signaux les attachent
// aux champs désignés (fieldTree) : ils apparaissent dans f.nom().errors(),
// comme les erreurs d'un validateur. Leur message est ici la CLÉ envoyée par
// le serveur, que le template traduit (ngx-translate). Essayez : un nom
// vide, un titre "Dr", un homonyme (Alice DURAND), un login déjà pris
// (martin), la même fiche modifiée dans deux onglets...
// =========================================================================

import { Component, computed, inject, signal } from '@angular/core';
import { FormField, form, submit, ValidationError } from '@angular/forms/signals';
import { TranslatePipe, TranslateService } from '@ngx-translate/core';
import { ApiService } from '../../../commun/api.service';
import { BarreConnexion } from '../../../commun/barre-connexion';
import { ErreurApi } from '../../../commun/erreur-api';
import { Personne, Utilisateur } from '../../../commun/types';
// la fenêtre de confirmation de l'exemple 16, réutilisée telle quelle
import { ConfirmationService } from '../../../16-confirmation/src/app/confirmation/confirmation.service';
import { FenetreConfirmation } from '../../../16-confirmation/src/app/confirmation/fenetre-confirmation';

interface SaisieClient {
  titre: string;
  prenom: string;
  nom: string;
  login: string;
  password: string;
  version: number;
}

const VIDE: SaisieClient = { titre: 'Mr', prenom: '', nom: '', login: '', password: '', version: 0 };

@Component({
  selector: 'app-root',
  imports: [FormField, TranslatePipe, BarreConnexion, FenetreConfirmation],
  templateUrl: './app.html',
})
export class App {
  private readonly api = inject(ApiService);
  private readonly confirmation = inject(ConfirmationService);
  protected readonly translate = inject(TranslateService);

  protected readonly utilisateur = signal<Utilisateur | null>(null);
  protected readonly clients = signal<Personne[]>([]);
  protected readonly estAdmin = computed(() => this.utilisateur()?.role === 'ADMIN');

  // le formulaire : idEdite === null -> création, sinon modification
  protected readonly idEdite = signal<number | null>(null);
  protected readonly modele = signal<SaisieClient>({ ...VIDE });
  protected readonly f = form(this.modele); // aucun validateur : c'est le serveur qui valide
  // le message général de la dernière erreur, et celui d'un succès
  protected readonly erreurPage = signal<ErreurApi | null>(null);
  protected readonly succes = signal<{ cle: string; params: Record<string, string> } | null>(null);

  protected async surConnexion(u: Utilisateur | null): Promise<void> {
    this.utilisateur.set(u);
    this.clients.set([]);
    if (u?.role === 'ADMIN') {
      await this.recharger();
    }
  }

  private async recharger(): Promise<void> {
    this.clients.set(await this.api.get<Personne[]>('/clients'));
  }

  protected nouveau(): void {
    this.idEdite.set(null);
    this.modele.set({ ...VIDE });
    this.f().reset(); // l'état des champs (et leurs erreurs) repart de zéro
    this.effacerMessages();
  }

  protected editer(c: Personne): void {
    this.idEdite.set(c.id);
    // la VERSION lue avec la fiche repart avec la modification : si quelqu'un
    // a modifié la fiche entre-temps, le serveur répondra 409 CONCURRENT_UPDATE
    this.modele.set({ titre: c.titre, prenom: c.prenom, nom: c.nom, login: c.login ?? '', password: '', version: c.version });
    this.f().reset();
    this.effacerMessages();
  }

  private effacerMessages(): void {
    this.erreurPage.set(null);
    this.succes.set(null);
  }

  // l'erreur de CHAQUE champ, renvoyée par le serveur -> une erreur du formulaire
  private erreursDesChamps(e: ErreurApi): ValidationError.WithOptionalFieldTree[] {
    const champs = Object.keys(this.modele()) as (keyof SaisieClient)[];
    return Object.entries(e.champs)
      .filter(([champ]) => champs.includes(champ as keyof SaisieClient))
      .map(([champ, cle]) => ({ kind: 'serveur', message: cle, fieldTree: this.f[champ as keyof SaisieClient] }));
  }

  protected async enregistrer(evenement: SubmitEvent): Promise<void> {
    evenement.preventDefault();
    this.effacerMessages();
    const id = this.idEdite();
    await submit(this.f, async () => {
      try {
        const corps = this.modele();
        const client =
          id === null
            ? await this.api.post<Personne>('/clients', corps)
            : await this.api.put<Personne>(`/clients/${id}`, corps);
        this.nouveau(); // formulaire vidé, messages effacés...
        // ... sauf celui-ci
        this.succes.set({
          cle: id === null ? 'FORM.AJOUTE' : 'FORM.MODIFIE',
          params: { nom: `${client.prenom} ${client.nom}`, login: client.login ?? '' },
        });
        await this.recharger();
        return null;
      } catch (e) {
        if (!(e instanceof ErreurApi)) {
          throw e;
        }
        this.erreurPage.set(e); // le message général
        return this.erreursDesChamps(e); // l'erreur de chaque champ
      }
    });
  }

  protected async supprimer(c: Personne): Promise<void> {
    const nom = `${c.prenom} ${c.nom}`;
    if (!(await this.confirmation.confirmer(this.translate.instant('FORM.CONFIRMER_SUPPRESSION', { nom })))) {
      return;
    }
    this.effacerMessages();
    try {
      await this.api.delete(`/clients/${c.id}`);
      this.succes.set({ cle: 'FORM.SUPPRIME', params: { nom } });
    } catch (e) {
      this.erreurPage.set(e instanceof ErreurApi ? e : null);
    }
    await this.recharger();
  }
}
