    // =========================================================================
// 01-premiere-application/src/app/app.ts - un premier composant
// -------------------------------------------------------------------------
// Un composant Angular est une CLASSE TypeScript, décorée par @Component :
//   - selector    : la balise HTML qui le représente (<app-root>) ;
//   - templateUrl : le fichier de son HTML (app.html) ;
//   - styleUrl    : le fichier de son CSS (app.css), qui ne s'applique
//                   qu'à ce composant.
// Les propriétés et méthodes de la classe sont utilisables dans le template.
// =========================================================================

import { Component, signal } from '@angular/core';

@Component({
  selector: 'app-root',
  templateUrl: './app.html',
  styleUrl: './app.css',
})
export class App {
  // des données "ordinaires" : le template peut les afficher
  protected readonly titre = 'Bienvenue dans Angular';
  protected readonly langages = ['TypeScript', 'HTML', 'CSS'];

  // signal(...) : une donnée RÉACTIVE. On lit sa valeur en l'APPELANT
  // (compteur()), on la change par set() ou update(). Angular met alors à
  // jour AUTOMATIQUEMENT les endroits de la page qui l'affichent.
  protected readonly compteur = signal(0);
  protected readonly heure = signal(new Date().toLocaleTimeString());

  // une méthode appelée par le bouton du template
  protected incrementer(): void {
    this.compteur.update((n) => n + 1);
    this.heure.set(new Date().toLocaleTimeString());
  }
}
    • lignes 14-18 : le décorateur [@Component] fait de la classe un composant :
      • ligne 15 : son sélecteur, la balise qui le représente dans un template ou dans [index.html] ;
      • ligne 16 : le fichier de son template HTML ;
      • ligne 17 : le fichier de son style CSS ;
    • lignes 21-22 : deux données ordinaires. [protected] : elles ne sont utilisées que par le composant et par son template ; [readonly] : elles ne seront pas remplacées ;
    • ligne 27 : [signal(0)] crée une donnée réactive, de valeur initiale 0. Un signal est une fonction : on lit sa valeur en l'appelant ([this.compteur()]), et on la change par [set(valeur)] ou [update(fonction)]. Quand elle change, [Angular] met à jour tous les endroits de la page qui l'affichent ;
    • lignes 31-34 : une méthode qui modifie deux signaux. Elle ne touche pas à la page : c'est [Angular] qui la mettra à jour. [update((n) => n + 1)] calcule la nouvelle valeur à partir de l'ancienne.

    // =========================================================================
// 02-signaux/src/app/app.ts - signal, computed, effect, linkedSignal
// -------------------------------------------------------------------------
// Un petit calculateur de prix TTC :
//   - signal       : une valeur réactive, qu'on lit en l'appelant et
//                    qu'on modifie par set() / update() ;
//   - computed     : une valeur CALCULÉE à partir d'autres signaux,
//                    recalculée automatiquement quand ils changent ;
//   - effect       : du code exécuté à chaque changement des signaux qu'il
//                    lit (un "effet de bord") ;
//   - linkedSignal : un signal modifiable, mais REMIS à une valeur calculée
//                    quand un autre signal change.
// =========================================================================

import { Component, computed, effect, linkedSignal, signal, untracked } from '@angular/core';
// FormsModule apporte [(ngModel)], qui relie un champ à un signal (cf. exemple 05)
import { FormsModule } from '@angular/forms';

@Component({
  selector: 'app-root',
  imports: [FormsModule],
  templateUrl: './app.html',
})
export class App {
  protected readonly libelle = signal('Stylo');
  protected readonly prixHT = signal(2.5);
  protected readonly tauxTVA = signal(20); // en %

  // linkedSignal : la quantité revient à 1 chaque fois que l'article change,
  // mais l'utilisateur peut la modifier entre-temps
  protected readonly quantite = linkedSignal({ source: this.libelle, computation: () => 1 });

  // computed : jamais affecté directement, toujours recalculé. Angular sait
  // de quels signaux il dépend (ceux qu'il LIT) : il le recalcule quand l'un
  // d'eux change, et seulement dans ce cas
  protected readonly totalHT = computed(() => this.quantite() * this.prixHT());
  protected readonly montantTVA = computed(() => (this.totalHT() * this.tauxTVA()) / 100);
  protected readonly totalTTC = computed(() => this.totalHT() + this.montantTVA());

  // un formateur de montants en euros, à la française (1 234,50 €)
  private readonly euros = new Intl.NumberFormat('fr-FR', { style: 'currency', currency: 'EUR' });

  // l'historique des totaux TTC, rempli par l'effet ci-dessous
  protected readonly historique = signal<string[]>([]);
  private ancienTTC: number | null = null;

  constructor() {
    // effect : exécuté une première fois, puis à chaque changement de
    // totalTTC (le seul signal qu'il lit)
    effect(() => {
      const ttc = this.totalTTC();
      document.title = `TTC : ${this.format(ttc)}`;
      if (this.ancienTTC !== null) {
        const ligne = `${this.format(this.ancienTTC)} -> ${this.format(ttc)}`;
        // untracked : modifier historique SANS en faire une dépendance de l'effet
        untracked(() => this.historique.update((h) => [ligne, ...h].slice(0, 5)));
      }
      this.ancienTTC = ttc;
    });
  }

  protected format(montant: number): string {
    return this.euros.format(montant);
  }

  protected reinitialiser(): void {
    this.libelle.set('Stylo');
    this.prixHT.set(2.5);
    this.tauxTVA.set(20);
    this.quantite.set(1);
  }
}
    • ligne 15 : les fonctions de réactivité, toutes fournies par [@angular/core] ;
    • lignes 21 et 17 : le template utilise [[(ngModel)]], qui vient du module [FormsModule]. Un composant doit importer (tableau [imports]) tout ce que son template utilise : les autres composants, les directives, les pipes. Le compilateur signale une erreur si l'on oublie ;

// 04-evenements/src/app/app.ts - réagir aux actions de l'utilisateur
// -------------------------------------------------------------------------
// (evenement)="instruction" : l'instruction est exécutée quand l'élément
// émet cet événement :
//   - (click)="f()"            appelle f ;
//   - (click)="f(3, $event)"   appelle f avec des arguments ; $event est
//                              l'événement natif du navigateur ;
//   - (keyup.enter), (keydown.escape) : seulement pour cette touche ;
//   - #nom sur un élément : une "variable de référence" du template, qui
//     désigne l'élément HTML lui-même (nom.value, nom.focus()...).
// Ici, pas de FormsModule ni de ngModel : on lit les champs nous-mêmes.

import { Component, signal } from '@angular/core';

interface Tache {
  id: number;
  libelle: string;
  faite: boolean;
}

@Component({
  selector: 'app-root',
  templateUrl: './app.html',
  styles: `
    .zone {
      background: #e9f2ff;
      height: 80px;
      cursor: crosshair;
    }
  `,
})
export class App {
  protected readonly taches = signal<Tache[]>([
    { id: 1, libelle: 'Installer Node.js', faite: true },
    { id: 2, libelle: 'Lire le chapitre sur les événements', faite: false },
  ]);
  protected readonly journal = signal<string[]>([]);
  private prochainId = 3;

  private tracer(message: string): void {
    this.journal.update((j) => [`${new Date().toLocaleTimeString()} ${message}`, ...j]);
  }

  // appelée par (submit) : $event.preventDefault() empêche le navigateur
  // d'envoyer le formulaire au serveur (et de recharger la page)
  protected ajouter(evenement: SubmitEvent, champ: HTMLInputElement): void {
    evenement.preventDefault();
    const libelle = champ.value.trim();
    if (libelle === '') {
      return;
    }
    // un signal contenant un tableau : on le REMPLACE par un nouveau tableau
    this.taches.update((t) => [...t, { id: this.prochainId++, libelle, faite: false }]);
    this.tracer(`ajout de « ${libelle} »`);
    champ.value = '';
    champ.focus();
  }

  protected basculer(tache: Tache, faite: boolean): void {
    this.taches.update((t) => t.map((x) => (x.id === tache.id ? { ...x, faite } : x)));
    this.tracer(`« ${tache.libelle} » : ${faite ? 'faite' : 'à faire'}`);
  }

  protected supprimer(tache: Tache): void {
    this.taches.update((t) => t.filter((x) => x.id !== tache.id));
    this.tracer(`suppression de « ${tache.libelle} »`);
  }

  // $event : l'événement natif (ici un MouseEvent), avec ses coordonnées
  protected position(evenement: MouseEvent): void {
    this.tracer(`clic en (${evenement.offsetX}, ${evenement.offsetY})`);
  }
}

// 06-formulaires-signaux/src/app/app.ts - les formulaires à signaux
// -------------------------------------------------------------------------
// La manière moderne (Angular 22) d'écrire un formulaire :
//   1. le MODÈLE : un signal qui contient un objet, avec une propriété par
//      champ (le formulaire lit et écrit dans ce signal) ;
//   2. form(modele, schema) : fabrique un ARBRE DE CHAMPS (FieldTree), qui
//      a la même forme que le modèle : f.nom, f.age, f.newsletter... Le
//      schéma (facultatif) décrit les règles des champs (cf. exemple 07) ;
//   3. [formField]="f.nom" relie un champ HTML à un champ de l'arbre.
// Chaque champ a un ÉTAT, obtenu en l'appelant : f.nom() donne sa valeur
// (value()), s'il a été touché (touched()), modifié (dirty()), s'il est
// désactivé (disabled())...
// =========================================================================

import { Component, signal } from '@angular/core';
import { JsonPipe } from '@angular/common';
import { disabled, form, FormField } from '@angular/forms/signals';

// la forme des données du formulaire
interface Profil {
  titre: string;
  prenom: string;
  nom: string;
  age: number;
  newsletter: boolean;
  email: string;
  pays: string;
  naissance: string;
  presentation: string;
}

const DEPART: Profil = {
  titre: 'Mme',
  prenom: 'Marie',
  nom: 'PELISSIER',
  age: 30,
  newsletter: false,
  email: '',
  pays: 'fr',
  naissance: '',
  presentation: '',
};

@Component({
  selector: 'app-root',
  imports: [FormField, JsonPipe],
  templateUrl: './app.html',
})
export class App {
  protected readonly titres = ['Mr', 'Mme', 'Melle'];

  // 1. le modèle
  protected readonly modele = signal<Profil>({ ...DEPART });

  // 2. l'arbre des champs ; une règle : l'e-mail n'est modifiable que si la
  // case "newsletter" est cochée (valueOf lit la valeur d'un AUTRE champ)
  protected readonly f = form(this.modele, (p) => {
    disabled(p.email, ({ valueOf }) => !valueOf(p.newsletter));
  });

  // les champs dont on affiche l'état, dans le tableau de droite
  protected readonly champs = ['prenom', 'nom', 'age', 'email'] as const;

  protected reinitialiser(): void {
    // on remet le modèle à sa valeur de départ, et l'état des champs à zéro
    this.modele.set({ ...DEPART });
    this.f().reset();
  }

  protected majuscules(): void {
    // modifier le MODÈLE par le code : les champs suivent
    this.modele.update((m) => ({ ...m, nom: m.nom.toUpperCase(), prenom: m.prenom.toUpperCase() }));
  }
}

// 07-validation/src/app/app.ts - valider un formulaire dans le navigateur
// -------------------------------------------------------------------------
// Les règles des champs sont écrites dans le SCHÉMA du formulaire (second
// argument de form()) : required, maxLength, minLength, pattern... et
// validate() pour une règle "maison". Chaque champ connaît alors ses
// erreurs (f.nom().errors()), recalculées à chaque frappe.
// On ne les affiche pas d'emblée (un formulaire vide couvert de rouge est
// désagréable) : seulement pour un champ déjà quitté ("touché"). submit()
// marque tous les champs comme touchés, puis n'exécute l'action que si le
// formulaire est valide.
//
// Mêmes règles que le serveur RdvMedecins (formulaire "Créer un compte").
// ATTENTION : cette validation n'est qu'un CONFORT pour l'utilisateur (une
// réponse immédiate, sans aller-retour avec le serveur). Elle ne protège
// rien : n'importe qui peut envoyer au serveur une requête fabriquée à la
// main (curl...). Le serveur revalide donc TOUJOURS tout (cf. exemple 25).

import { Component, signal } from '@angular/core';
import { form, FormField, maxLength, minLength, pattern, required, submit, validate } from '@angular/forms/signals';

interface Compte {
  titre: string;
  prenom: string;
  nom: string;
  login: string;
  password: string;
  confirmation: string;
}

const TITRES = ['Mr', 'Mme', 'Melle'];
const FORMAT_LOGIN = /^[a-z0-9._-]{1,30}$/;
const MOT_DE_PASSE_MIN = 6;

@Component({
  selector: 'app-root',
  imports: [FormField],
  templateUrl: './app.html',
})
export class App {
  protected readonly titres = TITRES;
  protected readonly modele = signal<Compte>({
    titre: 'Mr',
    prenom: '',
    nom: '',
    login: '',
    password: '',
    confirmation: '',
  });

  // le schéma : p désigne les CHEMINS des champs (p.nom, p.login...)
  protected readonly f = form(this.modele, (p) => {
    validate(p.titre, ({ value }) =>
      TITRES.includes(value()) ? null : { kind: 'titre', message: 'Choisissez un titre dans la liste.' },
    );
    required(p.prenom, { message: 'Le prénom est obligatoire.' });
    maxLength(p.prenom, 20, { message: 'Le prénom ne doit pas dépasser 20 caractères.' });
    required(p.nom, { message: 'Le nom est obligatoire.' });
    maxLength(p.nom, 20, { message: 'Le nom ne doit pas dépasser 20 caractères.' });
    required(p.login, { message: 'Le login est obligatoire.' });
    pattern(p.login, FORMAT_LOGIN, { message: 'Lettres minuscules non accentuées, chiffres, point, tiret, souligné.' });
    required(p.password, { message: 'Le mot de passe est obligatoire.' });
    minLength(p.password, MOT_DE_PASSE_MIN, { message: `Au moins ${MOT_DE_PASSE_MIN} caractères.` });
    // une règle "maison", qui compare DEUX champs (valueOf lit l'autre)
    validate(p.confirmation, ({ value, valueOf }) =>
      value() === valueOf(p.password)
        ? null
        : { kind: 'confirmation', message: 'La confirmation ne correspond pas au mot de passe.' },
    );
  });

  protected readonly envoiTente = signal(false);
  protected readonly succes = signal(false);

  // (submit) du formulaire
  protected async envoyer(evenement: SubmitEvent): Promise<void> {
    evenement.preventDefault();
    this.envoiTente.set(true);
    // submit() marque tous les champs comme touchés (leurs erreurs
    // apparaissent), puis n'exécute l'action que si tout est valide
    const valide = await submit(this.f, async () => {
      // ici, on enverrait le modèle au serveur (cf. exemple 25)
      return null; // null : pas d'erreur
    });
    this.succes.set(valide);
  }
}

// -------------------------------------------------------------------------
// Avant les formulaires à signaux (Angular 22), on écrivait les formulaires
// avec les FORMULAIRES RÉACTIFS (ReactiveFormsModule). On les rencontre dans
// la plupart des applications existantes : voici le formulaire de l'exemple
// 07, réécrit avec eux.
//   - un FormGroup regroupe des FormControl, un par champ ;
//   - chaque FormControl a une valeur de départ et des VALIDATEURS
//     (Validators.required, Validators.maxLength(20)...) ;
//   - dans le template, [formGroup] relie le <form> au groupe, et
//     formControlName="nom" relie un champ HTML à un contrôle ;
//   - l'état d'un contrôle (value, errors, touched, invalid...) est une
//     propriété ordinaire, et non un signal.
// =========================================================================

import { Component, inject, signal } from '@angular/core';
import { AbstractControl, FormBuilder, ReactiveFormsModule, ValidationErrors, Validators } from '@angular/forms';

const TITRES = ['Mr', 'Mme', 'Melle'];

// un validateur "maison" s'applique au GROUPE : il compare deux de ses champs.
// Il renvoie null (pas d'erreur) ou un objet décrivant l'erreur.
function confirmationIdentique(groupe: AbstractControl): ValidationErrors | null {
  const password = groupe.get('password')?.value;
  const confirmation = groupe.get('confirmation')?.value;
  return password === confirmation ? null : { confirmation: true };
}

@Component({
  selector: 'app-root',
  imports: [ReactiveFormsModule],
  templateUrl: './app.html',
})
export class App {
  protected readonly titres = TITRES;
  private readonly fb = inject(FormBuilder).nonNullable;

  // le groupe : { nom: [valeur de départ, validateurs] }
  protected readonly formulaire = this.fb.group(
    {
      titre: ['Mr', [Validators.required]],
      prenom: ['', [Validators.required, Validators.maxLength(20)]],
      nom: ['', [Validators.required, Validators.maxLength(20)]],
      login: ['', [Validators.required, Validators.pattern(/^[a-z0-9._-]{1,30}$/)]],
      password: ['', [Validators.required, Validators.minLength(6)]],
      confirmation: [''],
    },
    { validators: confirmationIdentique },
  );

  protected readonly succes = signal(false);
  protected readonly valeur = signal('');

  // un champ est signalé en erreur s'il a été touché et qu'il est invalide
  protected enErreur(nom: string): boolean {
    const controle = this.formulaire.get(nom);
    return !!controle && controle.touched && controle.invalid;
  }

  // (ngSubmit) : tous les champs deviennent "touchés", leurs erreurs apparaissent
  protected envoyer(): void {
    this.formulaire.markAllAsTouched();
    this.succes.set(this.formulaire.valid);
    // getRawValue() : la valeur de tout le groupe, typée
    this.valeur.set(JSON.stringify(this.formulaire.getRawValue()));
  }
}

// 09-pipes/src/app/pipes/horaire.pipe.ts - un pipe "maison"
// -------------------------------------------------------------------------
// Un pipe est une classe décorée par @Pipe, dont la méthode transform()
// reçoit la valeur placée AVANT la barre (et, éventuellement, des
// paramètres) et renvoie ce qu'il faut afficher :
//     {{ creneau | horaire }}          -> "8:00 - 8:20"
//     {{ creneau | horaire: 'debut' }} -> "8:00"
// Un pipe est "pur" par défaut : Angular ne rappelle transform() que si la
// valeur ou les paramètres ont CHANGÉ (même objet = même résultat).
// =========================================================================

import { Pipe, PipeTransform } from '@angular/core';

export interface Creneau {
  hdebut: number;
  mdebut: number;
  hfin: number;
  mfin: number;
}

// "8:05" plutôt que "8:5"
function heure(h: number, m: number): string {
  return `${h}:${String(m).padStart(2, '0')}`;
}

@Pipe({ name: 'horaire' })
export class HorairePipe implements PipeTransform {
  transform(c: Creneau, partie: 'tout' | 'debut' | 'fin' = 'tout'): string {
    switch (partie) {
      case 'debut':
        return heure(c.hdebut, c.mdebut);
      case 'fin':
        return heure(c.hfin, c.mfin);
      default:
        return `${heure(c.hdebut, c.mdebut)} - ${heure(c.hfin, c.mfin)}`;
    }
  }
}
    • ligne 27 : le nom du pipe, celui qu'on écrit dans les templates ;
    • ligne 29 : [transform(valeur, parametre)] renvoie ce qu'il faut afficher. Le paramètre est facultatif (valeur par défaut ['tout']) : [{{ creneau | horaire }}] affiche [8:00 - 8:20], [{{ creneau | horaire: 'debut' }}] affiche [8:00].

    Un pipe est pur par défaut : [Angular] ne rappelle [transform] que si la valeur ou l'un des paramètres a changé (un autre objet, une autre chaîne). C'est ce qui le rend plus efficace qu'une méthode du composant appelée dans le template (comme [format(...)] à l'exemple 02), qui est exécutée à chaque vérification du composant.
